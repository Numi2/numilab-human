ROOT = Path("/Users/n/numi-human-retained-delivery-20261009/skin-resting-multipose-clearance-1218/native-coupled-pair-preparation-011")
ANALYSIS = ROOT / "analysis-preparation-001"
ANALYZER = ANALYSIS / "analyze_matched_native_pair_011_revision_005.py"
ANALYZER_SHA256 = "9fb4cd3cc2479374a68caf8e5d38844aaff2a69dda454c7098a5eb41a71c8315"
BASELINE_LOG = ROOT / "baseline/native-run/native.log"
BASELINE_LOG_SHA256 = "edc959a6613fdf6c268bd2c86b5cfb0512bed4b3bb5cdbe3be7d4d0f38491338"
INTERVENTION_LOG = ROOT / "intervention/native-run/native.log"
INTERVENTION_LOG_SHA256 = "52046588aad19c75c9e2b01d1d5a8f87e9bc05593220867142c28149898a99a3"
BASELINE_INVOCATION = ROOT / "baseline/native-run/invocation.json"

def _sha(path: Path) -> str:
    return hashlib.sha256(path.read_bytes()).hexdigest()


def _load():
    assert _sha(ANALYZER) == ANALYZER_SHA256
    spec = importlib.util.spec_from_file_location("pair_analyzer_011_revision_005_test", ANALYZER)
    assert spec is not None and spec.loader is not None
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    return module


def _source_fingerprint(path: Path, expected_sha: str) -> str:
    assert _sha(path) == expected_sha
    matches = re.findall(r"resting_body_source_fingerprint=([0-9]+)", path.read_text(errors="replace"))
    assert len(matches) == 1
    return matches[0]


def test_real_011_pair_matches_exact_declared_source_append():
    analyzer = _load()
    baseline = _source_fingerprint(BASELINE_LOG, BASELINE_LOG_SHA256)
    intervention = _source_fingerprint(INTERVENTION_LOG, INTERVENTION_LOG_SHA256)
    result = analyzer.verify_intervention_body_source_fingerprint(
        baseline, intervention, (60.0, 100.0, 0.5))
    assert result["matched"] is True
    assert result["baseline_fingerprint_fnv64"] == "8518689296879886289"
    assert result["intervention_fingerprint_fnv64"] == "8303647443809329182"
    assert result["expected_intervention_fingerprint_fnv64"] == intervention
    assert result["payload_bytes_hex_little_endian_f64"] == (
        "0000000000004e400000000000005940000000000000e03f")
    base_inv = json.loads(BASELINE_INVOCATION.read_text())
    int_inv = json.loads(INTERVENTION_INVOCATION.read_text())
    assert _sha(BASELINE_INVOCATION) == BASELINE_INVOCATION_SHA256
    assert _sha(INTERVENTION_INVOCATION) == INTERVENTION_INVOCATION_SHA256
    base_decl = json.loads((ROOT / "baseline/run-declaration.json").read_text())
    int_decl = json.loads((ROOT / "intervention/run-declaration.json").read_text())
    assert "--drive-intervention" not in base_decl["argv"]
    assert int_decl["argv"].count("--drive-intervention") == 1
    di = int_decl["argv"].index("--drive-intervention")
    assert int_decl["argv"][di + 1:di + 4] == ["60.0", "100.0", "0.5"]
    assert "--resting-drive-intervention" not in base_inv["argv"]
    assert int_inv["argv"].count("--resting-drive-intervention") == 1
    ni = int_inv["argv"].index("--resting-drive-intervention")
    assert int_inv["argv"][ni + 1:ni + 4] == ["60.0", "100.0", "0.5"]

        analyzer.verify_intervention_body_source_fingerprint(
            "8518689296879886289", "8303647443809329182", (60.0, 101.0, 0.5))

def test_real_native_pair_argv_normalizes_exact_arm_local_outputs():
    analyzer = _load()
    assert _sha(BASELINE_INVOCATION) == BASELINE_INVOCATION_SHA256
    assert _sha(INTERVENTION_INVOCATION) == INTERVENTION_INVOCATION_SHA256
    baseline_argv = json.loads(BASELINE_INVOCATION.read_text())["argv"]
    intervention_argv = json.loads(INTERVENTION_INVOCATION.read_text())["argv"]
    result = analyzer.normalize_pair_commands(
        baseline_argv, intervention_argv,
        ROOT / "baseline/native-run", ROOT / "intervention/native-run",
        native=True)
    assert result["passed"] is True
    assert result["intervention_flag"] == "--resting-drive-intervention"


def test_wrong_native_movie_output_path_is_rejected():
    analyzer = _load()
    argv = json.loads(INTERVENTION_INVOCATION.read_text())["argv"]
    argv = list(argv)
    index = argv.index("--resting-movie") + 1
    argv[index] = str(ROOT / "baseline/native-run/native-viewer.mov")
    with pytest.raises(ValueError, match="movie output differs"):
        analyzer.normalize_arm_local_paths(
            argv, ROOT / "intervention/native-run", native=True)
