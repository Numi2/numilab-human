import importlib.util
import json
from pathlib import Path
import unittest

ANALYZER = Path('/Users/n/numi-human-retained-delivery-20261009/skin-resting-multipose-clearance-1218/native-coupled-pair-preparation-011/analysis-preparation-001/analyze_matched_native_pair_011_revision_003.py')
DECLARATION = Path('/Users/n/numi-human-retained-delivery-20261009/skin-resting-multipose-clearance-1218/native-coupled-pair-preparation-011/baseline/run-declaration.json')
INVOCATION = Path('/Users/n/numi-human-retained-delivery-20261009/skin-resting-multipose-clearance-1218/native-coupled-pair-preparation-011/baseline/native-run/invocation.json')
spec = importlib.util.spec_from_file_location('analyzer011_revision002_test', ANALYZER)
mod = importlib.util.module_from_spec(spec)
spec.loader.exec_module(mod)

class DeclaredObserverEnvironmentTests(unittest.TestCase):
    def test_actual_baseline_declaration_cli_matches_recorded_native_environment(self):
        declaration = json.loads(DECLARATION.read_text())
        invocation = json.loads(INVOCATION.read_text())
        declared = mod.verify_observer_env(mod.declared_environment_from_owner_argv(declaration['argv']))
        recorded = mod.verify_observer_env(invocation['environment'])
        self.assertEqual(declared, recorded)
        self.assertEqual(declared['accepted_geometry_capture_steps'], mod.CAPTURES)

    def test_missing_or_duplicate_tour_and_period_flags_are_rejected(self):
        bad = [
            ['--inspection-period-seconds', '2.0'],
            ['--inspection-tour'],
            ['--inspection-tour', '--inspection-tour', '--inspection-period-seconds', '2.0'],
            ['--inspection-tour', '--inspection-period-seconds', '2.0', '--inspection-period-seconds', '2.0'],
        ]
        for argv in bad:
            with self.subTest(argv=argv), self.assertRaises(ValueError):
                mod.declared_environment_from_owner_argv(list(argv))

    def test_wrong_period_and_conflicting_env_are_rejected(self):
        bad = [
            ['--inspection-tour', '--inspection-period-seconds', '2.5'],
            ['NUMI_HUMAN_RESTING_INSPECTION_TOUR=0', '--inspection-tour', '--inspection-period-seconds', '2.0'],
            ['NUMI_HUMAN_RESTING_INSPECTION_PERIOD_SECONDS=3.0', '--inspection-tour', '--inspection-period-seconds', '2.0'],
        ]
        for argv in bad:
            with self.subTest(argv=argv), self.assertRaises(ValueError):
                mod.declared_environment_from_owner_argv(argv)

if __name__ == '__main__':
    unittest.main()
