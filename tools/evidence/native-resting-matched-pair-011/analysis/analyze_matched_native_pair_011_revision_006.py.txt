PINNED_READS: dict[str, str] = {}


def sha(path: Path) -> str:
    h = hashlib.sha256()
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(1 << 20), b""):
            h.update(block)
    return h.hexdigest()


def pin_file(path: Path, expected: str | None = None) -> str:
    if path.is_symlink():
        raise ValueError(f"expected a regular, non-symlink input: {path}")
    path = path.resolve(strict=True)
    if not path.is_file():
        raise ValueError(f"expected a regular input: {path}")
    digest = sha(path)
    if expected is not None and digest != expected:
        raise ValueError(f"input hash changed: {path}: {digest} != {expected}")
    old = PINNED_READS.setdefault(str(path), digest)
    if old != digest:
        raise ValueError(f"input changed during analysis: {path}")
    return digest


def read_json(path: Path) -> dict[str, Any]:
    pin_file(path)
    value = json.loads(path.read_text(encoding="utf-8"))
    if not isinstance(value, dict):
        raise ValueError(f"expected JSON object: {path}")
    return value


def read_csv(path: Path, required: set[str] | None = None) -> tuple[list[str], list[dict[str, str]]]:
    pin_file(path)
    with path.open("r", encoding="utf-8", newline="") as stream:
        reader = csv.DictReader(stream)
        header = reader.fieldnames
        if not header or len(header) != len(set(header)):
            raise ValueError(f"missing or duplicate CSV header fields: {path}")
        missing = sorted((required or set()) - set(header))
        if missing:
            raise ValueError(f"{path} missing required columns: {missing}")
        rows = list(reader)
    if not rows or any(None in row or any(value is None for value in row.values()) for row in rows):
        raise ValueError(f"empty or malformed CSV rows: {path}")
    return header, rows


def finite(value: Any, label: str) -> float:
    try:
        result = float(value)
    except (TypeError, ValueError) as exc:
        raise ValueError(f"invalid {label}: {value!r}") from exc
    if not math.isfinite(result):
        raise ValueError(f"nonfinite {label}")
    return result


def float32_value(value: Any, label: str = "Float32 observation") -> float:
    number = finite(value, label)
    try:
        result = struct.unpack("<f", struct.pack("<f", number))[0]
    except (OverflowError, struct.error) as exc:
        raise ValueError(f"{label} is outside finite Float32 range") from exc
    if not math.isfinite(result):
        raise ValueError(f"{label} is outside finite Float32 range")
    return result


def float32_csv_value(token: str, label: str) -> float:
    result = float32_value(token, label)
    try:
        token_decimal = Decimal(token)
        roundtrip_decimal = Decimal(format(result, ".12g"))
    except InvalidOperation as exc:
        raise ValueError(f"{label} is not a valid decimal CSV token") from exc
    if not token_decimal.is_finite() or token_decimal != roundtrip_decimal:
        raise ValueError(f"{label} does not round-trip through the native 12-digit Float32 CSV format")
    return result


def float32_next(value: Any, direction: int) -> float:
    if direction not in (-1, 1):
        raise ValueError("Float32 neighbor direction must be -1 or +1")
    current = float32_value(value)
    bits = struct.unpack("<I", struct.pack("<f", current))[0]
    if current == 0.0:
        bits = 0x00000001 if direction > 0 else 0x80000001
    elif current > 0.0:
        bits += direction
    else:
        bits -= direction
    return struct.unpack("<f", struct.pack("<I", bits))[0]


def fraction_text(value: Fraction) -> str:
    return f"{value.numerator}/{value.denominator}"


def outward_float(value: Fraction, *, lower: bool) -> float:
    result = float(value)
    represented = Fraction.from_float(result)
    if lower and represented > value:
        result = math.nextafter(result, -math.inf)
    elif not lower and represented < value:
        result = math.nextafter(result, math.inf)
    return result


def float32_rounding_cell(token: str, label: str = "Float32 observation") -> tuple[Fraction, Fraction, Fraction]:
    """Return center and conservative midpoint hull, including ties at both ends."""
    center_value = float32_csv_value(token, label)
    previous = float32_next(center_value, -1)
    following = float32_next(center_value, 1)
    center = Fraction.from_float(center_value)
    if math.isfinite(previous):
        lower = (Fraction.from_float(previous) + center) / 2
    else:
        lower = center - (Fraction.from_float(following) - center) / 2
    if math.isfinite(following):
        upper = (center + Fraction.from_float(following)) / 2
    else:
        upper = center + (center - Fraction.from_float(previous)) / 2
    return center, lower, upper


def float32_mean_interval(tokens: list[str], label: str) -> dict[str, Any]:
    if not tokens:
        raise ValueError(f"no samples for Float32 interval mean: {label}")
    cells = [float32_rounding_cell(token, label) for token in tokens]
    count = len(cells)
    midpoint = sum((x[0] for x in cells), Fraction(0)) / count
    lower = sum((x[1] for x in cells), Fraction(0)) / count
    upper = sum((x[2] for x in cells), Fraction(0)) / count
    return {
        "sample_count": count,
        "midpoint_mmhg": float(midpoint),
        "interval_mmhg": {"lower": outward_float(lower, lower=True),
                           "upper": outward_float(upper, lower=False)},
        "exact_fraction": {"midpoint": fraction_text(midpoint),
                            "lower": fraction_text(lower),
                            "upper": fraction_text(upper)},
    }


def float32_gas_window_summaries(trace_path: Path, trace_rows: list[dict[str, float]],
                                 owner: Any) -> dict[str, Any]:
    _, raw_rows = read_csv(trace_path, {"time_s", "PaCO2_mmhg", "PaO2_mmhg"})
    if len(raw_rows) != len(trace_rows):
        raise ValueError("raw respiration CSV and owner trace reader returned different row counts")
    for index, (raw, parsed) in enumerate(zip(raw_rows, trace_rows)):
        if finite(raw["time_s"], "respiration time_s") != parsed["time_s"]:
            raise ValueError(f"raw respiration time differs from owner row {index}")
    summaries: dict[str, Any] = {}
    for name in ("pre", "dose"):
        start_s, end_s = WINDOWS[name]
        selected = [row for row in raw_rows
                    if start_s <= finite(row["time_s"], "respiration time_s") < end_s]
        owner_selected = owner.window_rows(trace_rows, start_s, end_s)
        if len(selected) != len(owner_selected):
            raise ValueError(f"{name} raw Float32 window does not match owner sample selection")
        summaries[name] = {
            "window_s": [start_s, end_s],
            "PaCO2_mmhg": float32_mean_interval(
                [row["PaCO2_mmhg"] for row in selected], f"{name} PaCO2"),
            "PaO2_mmhg": float32_mean_interval(
                [row["PaO2_mmhg"] for row in selected], f"{name} PaO2"),
        }
    return summaries


def gas_did_rounding_interval(control: dict[str, Any], intervention: dict[str, Any],
                              gas: str) -> dict[str, Any]:
    cpre = control["gas_direction_float32_windows"]["pre"][gas]["exact_fraction"]
    cdose = control["gas_direction_float32_windows"]["dose"][gas]["exact_fraction"]
    tpre = intervention["gas_direction_float32_windows"]["pre"][gas]["exact_fraction"]
    tdose = intervention["gas_direction_float32_windows"]["dose"][gas]["exact_fraction"]
    cp, cd = Fraction(cpre["midpoint"]), Fraction(cdose["midpoint"])
    tp, td = Fraction(tpre["midpoint"]), Fraction(tdose["midpoint"])
    cpl, cph = Fraction(cpre["lower"]), Fraction(cpre["upper"])
    cdl, cdh = Fraction(cdose["lower"]), Fraction(cdose["upper"])
    tpl, tph = Fraction(tpre["lower"]), Fraction(tpre["upper"])
    tdl, tdh = Fraction(tdose["lower"]), Fraction(tdose["upper"])
    midpoint = (td - tp) - (cd - cp)
    lower = (tdl - tph) - (cdh - cpl)
    upper = (tdh - tpl) - (cdl - cph)
    return {
        "midpoint_mmhg": float(midpoint),
        "interval_mmhg": {"lower": outward_float(lower, lower=True),
                           "upper": outward_float(upper, lower=False)},
        "exact_fraction": {"midpoint": fraction_text(midpoint),
                            "lower": fraction_text(lower),
                            "upper": fraction_text(upper)},
        "sample_counts": {
            "control_pre": control["gas_direction_float32_windows"]["pre"][gas]["sample_count"],
            "control_dose": control["gas_direction_float32_windows"]["dose"][gas]["sample_count"],
            "intervention_pre": intervention["gas_direction_float32_windows"]["pre"][gas]["sample_count"],
            "intervention_dose": intervention["gas_direction_float32_windows"]["dose"][gas]["sample_count"],
        },
        "representation_scope": "Float32 input rounding-cell hull only; excludes integration, physiology, and model uncertainty.",
    }


def integer(value: Any, label: str) -> int:
    number = finite(value, label)
    if not number.is_integer():
        raise ValueError(f"noninteger {label}: {value!r}")
    return int(number)


def canonical_sha(value: Any) -> str:
    raw = json.dumps(value, sort_keys=True, separators=(",", ":"), allow_nan=False).encode()
    return hashlib.sha256(raw).hexdigest()


def source_block_sha256(path: Path, token: str) -> str:
    text = path.read_text(encoding="utf-8")
    start = text.index(token)
    brace = text.index("{", start)
    depth = 0
    for index in range(brace, len(text)):
        if text[index] == "{":
            depth += 1
        elif text[index] == "}":
            depth -= 1
            if depth == 0:
                return hashlib.sha256(text[start:index + 1].encode("utf-8")).hexdigest()
    raise ValueError(f"unterminated source block: {token}")

def accepted_runtime_guard_passed(identity_verified: bool, accepted_steps: int,
                                   expected_steps: int = STEPS) -> bool:
    # A complete horizon plus exact source/runtime identity means each committed
    # candidate passed the source status guard. This does not cover rejected
    # trial candidates or provide an independently normalized residual.
    return identity_verified is True and accepted_steps == expected_steps

    try:
        fixture_sha = pin_file(GAS_SOURCE_FIXTURE)
    except (OSError, ValueError) as exc:
        fixture_sha = f"unavailable:{type(exc).__name__}"

        build_rule_text = GAS_BUILD_RULE.read_text(encoding="utf-8")

        build_rule_sha, build_rule_matches = "unavailable", False
    ref = declaration.get("runtime_launcher_chain", {}).get("build_provenance", {})

        "resolve": "kernel void nm_human_respiration_resolve(",

def verify_accepted_grid(rows: list[dict[str, float]], *, steps: int = STEPS,
                         dt: float = DT, stride: int = 8) -> dict[str, Any]:
    if steps <= 0 or stride <= 0 or steps % stride or not math.isfinite(dt) or dt <= 0:
        raise ValueError("invalid declared accepted grid")
    expected = tuple(range(stride, steps + 1, stride))
    actual: list[tuple[int, float]] = []
    previous = 0
    for idx, row in enumerate(rows):
        step_value = finite(row.get("step"), f"step row {idx}")
        if not step_value.is_integer():
            raise ValueError(f"noninteger accepted step at row {idx}")
        step = int(step_value)
        if step <= previous:
            raise ValueError(f"duplicate or out-of-order accepted step at row {idx}")
        if step > steps or step % stride:
            raise ValueError(f"accepted step outside declared cadence at row {idx}")
        time_s = finite(row.get("time_s"), f"time_s row {idx}")
        expected_time = float(format(step * struct.unpack("<f", struct.pack("<f", dt))[0], ".12g"))
        if time_s != expected_time:
            raise ValueError(f"accepted time shifted from step*Float32(dt) at step {step}")
        actual.append((step, time_s))
        previous = step
    got = tuple(step for step, _ in actual)
    if got != expected:
        raise ValueError("accepted grid is incomplete or shifted")
    return {"passed": True, "row_count": len(actual), "step_stride": stride,
            "first_step": actual[0][0], "last_step": actual[-1][0],
            "first_time_s": actual[0][1], "last_time_s": actual[-1][1],
            "sequence_sha256": canonical_sha(actual)}


def _row_step(row: dict[str, str], label: str) -> int:
    for key in ("step", "accepted_step"):
        if key in row:
            return integer(row[key], f"{label}.{key}")
    raise ValueError(f"{label} lacks step identity")


def exact_pre_dose_rows(control: list[dict[str, str]], treatment: list[dict[str, str]],
                        *, step_key: str = "step") -> dict[str, Any]:
    """Require raw-cell equality before the declared 60 s intervention start."""
    left = [r for r in control if integer(r.get(step_key), step_key) < DOSE_FIRST_STEP]
    right = [r for r in treatment if integer(r.get(step_key), step_key) < DOSE_FIRST_STEP]
    if not left or len(left) != len(right):
        raise ValueError("pre-dose samples are missing or differ in count")
    if left != right:
        mismatch = next((i for i, (a, b) in enumerate(zip(left, right)) if a != b), None)
        raise ValueError(f"pre-dose measured rows differ at filtered row {mismatch}")
    steps = [_row_step(row, "pre-dose row") for row in left]
    return {"passed": True, "exact_raw_rows": len(left), "first_step": steps[0],
            "last_step": steps[-1], "rows_sha256": canonical_sha(left),

FAILURE_RECEIPT_ENV = "NUMI_HUMAN_RESTING_COMMON_FAILURE_RECEIPT"


def normalize_arm_local_paths(argv: list[str], run_dir: Path, *, native: bool = False) -> list[str]:
    """Normalize only the declared run output and failure-receipt paths."""
    result = list(argv)
    expected_run = str(run_dir.resolve())
    if native:
        if len(result) < 5 or str(Path(result[4]).resolve()) != expected_run:
            raise ValueError("native argv positional run output differs from its arm directory")
        result[4] = "<ARM_NATIVE_RUN>"
        movie_indices = [i for i, token in enumerate(result) if token == "--resting-movie"]
        if len(movie_indices) != 1 or movie_indices[0] + 1 >= len(result):
            raise ValueError("native argv must contain exactly one complete --resting-movie path")
        movie_index = movie_indices[0] + 1
        expected_movie = expected_run + "/native-viewer.mov"
        if result[movie_index] != expected_movie:
            raise ValueError("native argv movie output differs from its arm directory")
        result[movie_index] = "<ARM_NATIVE_RUN>/native-viewer.mov"
        return result

    output_indices = [i for i, token in enumerate(result) if token == "--output"]
    if len(output_indices) != 1 or output_indices[0] + 1 >= len(result):
        raise ValueError("owner argv must contain exactly one complete --output path")
    output_index = output_indices[0] + 1
    if result[output_index] != expected_run:
        raise ValueError("owner argv --output path differs from its arm directory")
    result[output_index] = "<ARM_NATIVE_RUN>"

    receipt_indices = [i for i, token in enumerate(result) if token.startswith(FAILURE_RECEIPT_ENV + "=")]
    expected_receipt = expected_run + "/common-field-failure.json"
    if len(receipt_indices) != 1:
        raise ValueError("owner argv must contain exactly one common failure-receipt path")
    receipt_index = receipt_indices[0]
    key, value = result[receipt_index].split("=", 1)
    if key != FAILURE_RECEIPT_ENV or value != expected_receipt:
        raise ValueError("owner argv common failure-receipt path differs from its arm directory")
    result[receipt_index] = key + "=<ARM_NATIVE_RUN>/common-field-failure.json"
    return result


def normalize_arm_local_environment(environment: dict[str, str], run_dir: Path) -> dict[str, str]:
    result = dict(environment)
    expected = str(run_dir.resolve()) + "/common-field-failure.json"
    if result.get(FAILURE_RECEIPT_ENV) != expected:
        raise ValueError("native environment common failure-receipt path differs from its arm directory")
    result[FAILURE_RECEIPT_ENV] = "<ARM_NATIVE_RUN>/common-field-failure.json"
    return result


def normalize_pair_environments(control: dict[str, str], treatment: dict[str, str],
                                control_run: Path, treatment_run: Path) -> dict[str, Any]:
    left = normalize_arm_local_environment(control, control_run)
    right = normalize_arm_local_environment(treatment, treatment_run)
    if left != right:
        raise ValueError("native environments differ beyond the declared common failure-receipt path")
    return {"passed": True, "normalized_baseline_environment_sha256": canonical_sha(left),
            "normalized_intervention_environment_sha256": canonical_sha(right)}


def _remove_intervention(argv: list[str], flag: str, present: bool) -> list[str]:
    result = list(argv)
    indices = [i for i, token in enumerate(result) if token == flag]
    if present:
        if len(indices) != 1 or indices[0] + 3 >= len(result):
            raise ValueError(f"expected exactly one {flag} with three values")
        i = indices[0]
        values = result[i + 1:i + 4]
        if values != ["60.0", "100.0", "0.5"]:
            raise ValueError(f"{flag} is not the declared 60–100 s, 0.5 scaling")
        del result[i:i + 4]
    elif indices:
        raise ValueError(f"control unexpectedly contains {flag}")
    return result


def normalize_pair_commands(control: list[str], treatment: list[str],
                             control_run: Path, treatment_run: Path,

        raise ValueError("owner declaration must specify one inspection-tour flag and period")
    period_index = period_flags[0]
    if period_index + 1 >= len(argv) or argv[period_index + 1] != "2.0":

def _fnv64_payload(payload: bytes) -> int:
    value = FNV64_OFFSET
    for byte in payload:
        value = ((value ^ byte) * FNV64_PRIME) & 0xFFFFFFFFFFFFFFFF
    return value or FNV64_OFFSET


def _append_payload_owner_fingerprint(source_fingerprint: int, domain: str,
                                      payload_fingerprint: int) -> int:
    source_fingerprint = int(source_fingerprint)
    source_fingerprint ^= _fnv64_payload(domain.encode("utf-8"))
    source_fingerprint = (source_fingerprint * FNV64_PRIME) & 0xFFFFFFFFFFFFFFFF
    source_fingerprint ^= int(payload_fingerprint)
    source_fingerprint = (source_fingerprint * FNV64_PRIME) & 0xFFFFFFFFFFFFFFFF
    return source_fingerprint or FNV64_OFFSET


def verify_intervention_body_source_fingerprint(baseline_fingerprint: Any,
                                                intervention_fingerprint: Any,
                                                intervention_values: tuple[float, ...]) -> dict[str, Any]:
    """Verify the declared intervention's exact source-owner append to baseline identity."""
    if sys.byteorder != "little":
        raise ValueError("runtime source identity tuple encoding is pinned to little-endian Apple bytes")
    try:
        values = tuple(float(value) for value in intervention_values)
        baseline_value = int(baseline_fingerprint)
        observed_value = int(intervention_fingerprint)
    except (TypeError, ValueError, OverflowError) as exc:
        raise ValueError("invalid paired body-source fingerprint inputs") from exc
    if values != (60.0, 100.0, 0.5):
        raise ValueError("intervention tuple differs from the declared 60-100 s, 0.5 source append")
    payload = struct.pack("<3d", *values)
    payload_hash = _fnv64_payload(payload)
    expected = _append_payload_owner_fingerprint(
        baseline_value, "resting_drive_intervention", payload_hash)
    if observed_value == baseline_value:
        raise ValueError("intervention incorrectly reuses the baseline body-source fingerprint")
    if observed_value != expected:
        raise ValueError("intervention body-source fingerprint does not match the declared source-owner append")
    return {
        "relation": "intervention_is_exact_declared_source_owner_append_to_baseline",
        "baseline_fingerprint_fnv64": str(baseline_value),
        "intervention_fingerprint_fnv64": str(observed_value),
        "expected_intervention_fingerprint_fnv64": str(expected),
        "matched": True,
        "owner_domain": "resting_drive_intervention",
        "intervention_values": list(values),
        "payload_bytes_hex_little_endian_f64": payload.hex(),
        "payload_fingerprint_fnv64": str(payload_hash),
        "source": {
            "application_source_path": str(RUNTIME_IDENTITY_APP_SOURCE),
            "application_source_sha256": RUNTIME_IDENTITY_APP_SOURCE_SHA256,
            "runtime_identity_helper_path": str(RUNTIME_IDENTITY_HELPER_SOURCE),
            "runtime_identity_helper_sha256": RUNTIME_IDENTITY_HELPER_SOURCE_SHA256,
            "rule": "The app appends the FNV64 fingerprint of the exact 3-double intervention payload under the resting_drive_intervention domain to baseline source identity.",
        },
    }

def expected_support_steps(steps: int, q_first: int, q_last: int, stride: int = 8) -> list[int]:
    if steps <= 0 or stride <= 0 or q_first <= 1 or q_last < q_first or q_last > steps:
        raise ValueError("invalid support/Q observer grid")
    samples = set(range(stride, steps + 1, stride))
    samples.update(range(q_first - 1, q_last + 1))
    return sorted(samples)


def validate_support_sample_rows(rows: list[dict[str, str]], *, steps: int = STEPS,
                                 q_first: int = Q_FIRST, q_last: int = Q_LAST,
                                 stride: int = 8) -> dict[str, Any]:
    expected = expected_support_steps(steps, q_first, q_last, stride)
    actual = [integer(row.get("accepted_step"), "support.accepted_step") for row in rows]
    if actual != expected:
        raise ValueError("COM/support diagnostic does not cover the declared union of stride-8 and Q-window samples")
    previous = 0
    statuses: dict[str, int] = {}
    for row, step in zip(rows, expected):
        sample_start = integer(row.get("sample_start_step"), "support.sample_start_step")
        segment = integer(row.get("segment_steps"), "support.segment_steps")
        if sample_start != previous or segment != step - previous:
            raise ValueError(f"COM/support segment does not start at prior emitted sample for step {step}")
        time_s = finite(row.get("time_s"), f"support.time_s at step {step}")
        if abs(time_s - step * FLOAT32_DT) > 1e-9:
            raise ValueError(f"COM/support time differs from the Float32 accepted clock at step {step}")
        status = row.get("momentum_residual_status", "").strip()
        if not status:
            raise ValueError(f"COM/support residual status is missing at step {step}")
        statuses[status] = statuses.get(status, 0) + 1
        previous = step
    return {"passed": True, "sample_count": len(rows), "first_step": actual[0],
            "last_step": actual[-1], "normal_stride_steps": stride,
            "q_window_steps": [q_first, q_last],
            "extra_predecessor_step": q_first - 1,
            "momentum_residual_status_counts": statuses,
            "momentum_residual_status_is_descriptive": True}


def validate_support_impulse_rows(rows: list[dict[str, str]], sample_rows: list[dict[str, str]], *,
                                  contact_count: int = 32, steps: int = STEPS,
                                  q_first: int = Q_FIRST, q_last: int = Q_LAST,
                                  stride: int = 8) -> dict[str, Any]:
    expected = expected_support_steps(steps, q_first, q_last, stride)
    if not sample_rows or [integer(r.get("accepted_step"), "support.accepted_step") for r in sample_rows] != expected:
        raise ValueError("support impulse join lacks the complete expected COM sample grid")
    expected_steps = [step for step in expected for _ in range(contact_count)]
    actual_steps = [integer(row.get("accepted_step"), "impulse.accepted_step") for row in rows]
    if actual_steps != expected_steps:
        raise ValueError("support impulses have missing, duplicate, or out-of-order contact rows")
    com_by_step = {integer(row["accepted_step"], "support.accepted_step"): row for row in sample_rows}
    for i, step in enumerate(expected):
        group = rows[i * contact_count:(i + 1) * contact_count]
        contact_indices = [integer(row.get("contact_index"), "impulse.contact_index") for row in group]
        if contact_indices != list(range(contact_count)):
            raise ValueError(f"support contact indices are not a unique ordered 0..{contact_count - 1} set at step {step}")
        identities = []
        for row in group:
            identity = (integer(row.get("contact_index"), "impulse.contact_index"),
                        integer(row.get("body_index"), "impulse.body_index"),
                        integer(row.get("source_geometry_index"), "impulse.source_geometry_index"))
            identities.append(identity)
            com = com_by_step[step]
            if (integer(row.get("sample_start_step"), "impulse.sample_start_step") != integer(com["sample_start_step"], "COM.sample_start_step") or
                    integer(row.get("segment_steps"), "impulse.segment_steps") != integer(com["segment_steps"], "COM.segment_steps")):
                raise ValueError(f"support impulse segment does not join the COM sample at step {step}")
            for key, value in row.items():
                if key in ("contact_index", "body_index", "source_geometry_index",
                           "accepted_step", "sample_start_step", "segment_steps"):
                    integer(value, f"support.{key}")
                else:
                    finite(value, f"support.{key}")
        if len(set(identities)) != contact_count:
            raise ValueError(f"support contact identity is duplicated at step {step}")
    return {"passed": True, "row_count": len(rows), "contact_rows_per_sample": contact_count,
            "sample_count": len(expected), "contact_index_ordered_complete": True,
            "joined_to_com_segments": True,
            "scope": "Per-contact impulses at each emitted COM observation. Normal cadence is stride 8; the bounded Q window adds its predecessor and every accepted Q step. These rows are not per-step outside that window."}


def support_summary(rows: list[dict[str, str]]) -> dict[str, Any]:
    grid = validate_support_sample_rows(rows)
    post = [r for r in rows if 10.0 <= finite(r["time_s"], "support.time_s") < STEPS * FLOAT32_DT]
    if len(post) < 100:
        raise ValueError("COM/support diagnostic lacks post-initialization samples")
    xyz = [[finite(row[f"com_{axis}_m"], f"com_{axis}_m") for row in post] for axis in "xyz"]
    force = [finite(row["normal_force_last_physical_step_n"], "support_force") for row in post]
    contacts = [integer(row["active_contact_count"], "active_contact_count") for row in post]
    return {"sampled_rows_present": True, "sample_count": len(post), "full_observer_rows": len(rows),
            "sample_grid": grid,
            "com_endpoint_delta_mm_xyz": [1000.0 * (axis[-1] - axis[0]) for axis in xyz],
            "sampled_normal_force_last_physical_step_n": {"min": min(force), "mean": statistics.fmean(force), "max": max(force)},
            "sampled_active_contact_count": {"min": min(contacts), "mean": statistics.fmean(contacts), "max": max(contacts)},
            "momentum_residual_status_is_descriptive": True,
            "interpretation": "COM/support rows include the normal stride-8 grid and the Q-window predecessor plus every Q-window accepted step. Momentum residual status is a string category, not an integer pass bit. Sampled last-physical-step support force does not prove static equilibrium or momentum closure."}

def load_owner() -> Any:
    if pin_file(OWNER) != OWNER_SHA256:
        raise ValueError("pinned physiology owner changed")
    spec = importlib.util.spec_from_file_location("resting_intervention_study_1271_for_011", OWNER)
    if spec is None or spec.loader is None:
        raise ValueError("cannot load pinned physiology owner")
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    return module


def cli_output(argv: list[str]) -> Path:
    if argv.count("--output") != 1:
        raise ValueError("run declaration must contain exactly one --output")
    i = argv.index("--output")
    if i + 1 >= len(argv):
        raise ValueError("run declaration --output lacks a value")
    output = Path(argv[i + 1])
    if output.is_symlink():
        raise ValueError("run declaration output path is a symlink")
    return output.resolve(strict=True)

def verify_captures(run_dir: Path, invocation: dict[str, Any], log: str, declaration: dict[str, Any]) -> list[dict[str, Any]]:
    entries = []
    export_lines = [line for line in log.splitlines() if line.startswith("accepted_geometry_export=")]
    for step in CAPTURES:
        pack = run_dir / "accepted-geometry" / f"step-{step}.mrvpack"
        receipt_path = run_dir / "accepted-geometry" / f"step-{step}.receipt.json"
        pack_sha = pin_file(pack)
        receipt_sha = pin_file(receipt_path)
        matches = [line for line in export_lines if line.startswith(f"accepted_geometry_export={pack} ")]
        if len(matches) != 1:
            raise ValueError(f"native log does not uniquely bind accepted geometry step {step}")
        fields = dict(token.split("=", 1) for token in matches[0].split()[1:] if "=" in token)
        receipt = read_json(receipt_path)
        if (receipt.get("schema") != "numi.human.accepted-render-geometry.v1" or
                receipt.get("accepted_step") != step or receipt.get("physical_endpoint") != "accepted" or
                receipt.get("surface_audit_endpoint") != "passed" or
                receipt.get("accepted_pack_path") != str(pack) or receipt.get("pack_file_sha256") != pack_sha or
                fields.get("pack_sha256") != pack_sha or fields.get("receipt_sha256") != receipt_sha):
            raise ValueError(f"accepted geometry receipt/export mismatch at step {step}")
        accepted_time = finite(receipt.get("accepted_time_s"), "capture accepted_time_s")
        expected_time = step * FLOAT32_DT
        if abs(accepted_time - expected_time) > 1e-8:
            raise ValueError(f"capture accepted time differs from Float32 step clock at {step}")
        entries.append({"accepted_step": step, "nominal_time_s": step * DT, "accepted_time_s": accepted_time,
                        "pack_path": str(pack), "pack_sha256": pack_sha,
                        "receipt_path": str(receipt_path), "receipt_sha256": receipt_sha})
    if [entry["accepted_step"] for entry in entries] != declaration["capture_steps"]:
        raise ValueError("run captures differ from the exact declared step sequence")
    return entries


def trace_descriptives(trace_path: Path, owner: Any) -> dict[str, Any]:
    rows = owner.read_trace(trace_path)
    def stats(key: str) -> dict[str, float]:
        values = [finite(row[key], key) for row in rows]
        return {"min": min(values), "mean": statistics.fmean(values), "max": max(values)}
    return {"accepted_samples": len(rows), "first_last_time_s": [rows[0]["time_s"], rows[-1]["time_s"]],
            "blood_volume_ml": stats("blood_ml"), "PaCO2_mmhg": stats("PaCO2_mmhg"),
            "PaO2_mmhg": stats("PaO2_mmhg"), "SaO2_fraction": stats("SaO2"),
            "lung_volume_ml": stats("lung_volume_ml"), "airflow_ml_s": stats("airflow_ml_s"),
            "blood_error_abs_max_ml": max(abs(row["blood_error_ml"]) for row in rows),
            "oxygen_balance_error_abs_max_STPD_ml": max(abs(row["oxygen_balance_error_stpd_ml"]) for row in rows),
            "co2_balance_error_abs_max_STPD_ml": max(abs(row["co2_balance_error_stpd_ml"]) for row in rows),
            "cumulative_aortic_ejected_ml": rows[-1]["aortic_ejected_ml"] - rows[0]["aortic_ejected_ml"],
            "cumulative_pulmonary_ejected_ml": rows[-1]["pulmonary_ejected_ml"] - rows[0]["pulmonary_ejected_ml"],
            "complete_filling_ejection_cycles": rows[-1]["complete_filling_ejection_cycles"],
            "last_lv_stroke_volume_ml": rows[-1]["last_lv_stroke_ml"]}


def paired_response(control: dict[str, Any], intervention: dict[str, Any]) -> dict[str, Any]:
    c, t = control["owner_observation"], intervention["owner_observation"]
    dose_ve_delta = t["inspiratory_minute_ventilation_dose_L_min"] - c["inspiratory_minute_ventilation_dose_L_min"]
    did = t["primary_delta_PaCO2_mmhg"] - c["primary_delta_PaCO2_mmhg"]
    pco2_did = gas_did_rounding_interval(control, intervention, "PaCO2_mmhg")
    pao2_did = gas_did_rounding_interval(control, intervention, "PaO2_mmhg")
    recovery_pco2 = t["PaCO2_recovery_mean_mmhg"] - c["PaCO2_recovery_mean_mmhg"]
    recovery_pao2 = t["PaO2_recovery_mean_mmhg"] - c["PaO2_recovery_mean_mmhg"]
    c_ve = c["inspiratory_minute_ventilation_recovery_L_min"]
    t_ve = t["inspiratory_minute_ventilation_recovery_L_min"]
    if c_ve <= 0:
        raise ValueError("control recovery ventilation is nonpositive; relative recovery comparison undefined")
    relative_recovery_ve = abs(t_ve - c_ve) / c_ve
    pco2_lower, pco2_upper = (Fraction(pco2_did["exact_fraction"][key]) for key in ("lower", "upper"))
    pao2_lower, pao2_upper = (Fraction(pao2_did["exact_fraction"][key]) for key in ("lower", "upper"))
    if pco2_lower > 0:
