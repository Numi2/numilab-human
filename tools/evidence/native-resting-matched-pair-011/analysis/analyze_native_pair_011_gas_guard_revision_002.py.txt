def sha(path: Path) -> str:
 h=hashlib.sha256()
 with path.open("rb") as f:
  for b in iter(lambda:f.read(1<<20),b""): h.update(b)
 return h.hexdigest()

def pin(path: Path, expected: str|None=None) -> str:
 if path.is_symlink() or not path.is_file(): raise ValueError(f"not a regular file: {path}")
 got=sha(path)
 if expected is not None and got!=expected: raise ValueError(f"SHA mismatch: {path}")
 return got

def loadj(path: Path) -> dict[str,Any]:
 pin(path); obj=json.loads(path.read_text())
 if not isinstance(obj,dict): raise ValueError(f"expected JSON object: {path}")
 return obj

def invocation_matches_preview(declaration: dict[str,Any], invocation: dict[str,Any]) -> bool:
 preview=declaration.get("owner_cli_preview",{})

def final_guard_passed(identity: bool, accepted_steps: int, expected_steps: int,
                       execution_code: int, changed_inputs: dict[str,Any]) -> bool:

 ap.add_argument("--output",type=Path)

 output=args.output or (OUTPUT_DIR/f"gas-guard-{args.arm}.json")
 if output.exists() or output.is_symlink(): raise FileExistsError(output)
 result=analyze(args.arm)
 output.write_text(json.dumps(result,indent=2,sort_keys=True,allow_nan=False)+"\n")
 print(output)
 return 0 if result["status"]=="pass" else 1
