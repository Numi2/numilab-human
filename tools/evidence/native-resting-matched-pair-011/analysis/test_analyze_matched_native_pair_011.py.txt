def grid(steps, overrides=None):
    overrides = overrides or {}
    dt = struct.unpack("<f", struct.pack("<f", module.DT))[0]
    return [{"step": float(step), "time_s": overrides.get(step, float(format(step * dt, ".12g")))} for step in steps]


def make_q_index_map():
    q_indices = [i for i in range(129) if i not in range(3, 7)]
    rows = []
    for v, q in enumerate(q_indices):
        rows.append({"record_kind":"scalar_dof","local_q_index":str(q),"global_q_index":str(q),
                     "local_v_index":str(v),"global_v_index":str(v),"joint_index":"0",
                     "joint_name":"test","dof_name":str(v),"q_index_valid":"1"})
    for q in range(3, 7):
        rows.append({"record_kind":"configuration_without_direct_velocity","local_q_index":str(q),
                     "global_q_index":str(q),"local_v_index":"","global_v_index":"",
                     "joint_index":"","joint_name":"","dof_name":"","q_index_valid":"0"})
    for v in range(125,128):
        rows.append({"record_kind":"velocity_only","local_q_index":"","global_q_index":"",
                     "local_v_index":str(v),"global_v_index":str(v),"joint_index":"0",
                     "joint_name":"test","dof_name":str(v),"q_index_valid":"0"})
    return rows


class AcceptedGridTests(unittest.TestCase):
    def test_complete_exact_grid_passes(self):
        result = module.verify_accepted_grid(grid([8, 16, 24]), steps=24)
        self.assertTrue(result["passed"])
        self.assertEqual(result["last_step"], 24)

    def test_duplicate_or_missing_grid_fails(self):
        with self.assertRaisesRegex(ValueError, "duplicate or out-of-order"):
            module.verify_accepted_grid(grid([8, 16, 16, 24]), steps=24)
        with self.assertRaisesRegex(ValueError, "incomplete or shifted"):
            module.verify_accepted_grid(grid([8, 24]), steps=24)

    def test_shifted_time_fails(self):
        with self.assertRaisesRegex(ValueError, "shifted"):
            module.verify_accepted_grid(grid([8, 16, 24], {16: 0.1}), steps=24)


def constant_window(value, label):
    f32 = module.float32_value(value, label)
    token = format(f32, ".12g")
    return module.float32_mean_interval([token] * 100, label)

    def test_response_checks_require_resolved_expected_gas_directions(self):
        def arm(primary, ve_dose, pco2_rec, pao2_rec, ve_rec, pao2_dose_delta=0.0):
            return {
                "owner_observation": {
                    "primary_delta_PaCO2_mmhg": primary,
                    "inspiratory_minute_ventilation_dose_L_min": ve_dose,
                    "PaCO2_recovery_mean_mmhg": pco2_rec,
                    "PaO2_recovery_mean_mmhg": pao2_rec,
                    "inspiratory_minute_ventilation_recovery_L_min": ve_rec,
                    "PaCO2_dose_mean_mmhg": 40.0 + primary,
                    "PaO2_dose_mean_mmhg": 95.0 + pao2_dose_delta,
                    "SaO2_dose_mean": 0.97,
                },
                "gas_direction_float32_windows": {
                    "pre": {"PaCO2_mmhg": constant_window(40.0, "PaCO2"),
                            "PaO2_mmhg": constant_window(95.0, "PaO2")},
                    "dose": {"PaCO2_mmhg": constant_window(40.0 + primary, "PaCO2"),
                             "PaO2_mmhg": constant_window(95.0 + pao2_dose_delta, "PaO2")},
                },
            }
        control = arm(0.0, 8.0, 40.0, 95.0, 8.0, 0.0)
        treatment = arm(1.0, 7.0, 40.5, 94.0, 8.4, -1.0)
        result = module.paired_response(control, treatment)
        self.assertTrue(result["all_declared_response_checks_met"])
        self.assertEqual(len(result["predeclared_direction_and_recovery_checks"]), 6)
        self.assertGreater(result["values"]["float32_midpoint_PaCO2_difference_in_differences_mmhg"], 0.0)
        self.assertLess(result["values"]["float32_midpoint_PaO2_difference_in_differences_mmhg"], 0.0)

        zero = module.paired_response(control, arm(0.0, 7.0, 40.0, 95.0, 8.0, 0.0))
        self.assertFalse(zero["predeclared_direction_and_recovery_checks"][1]["passed"])
        self.assertFalse(zero["predeclared_direction_and_recovery_checks"][2]["passed"])
        self.assertIn("unresolved", zero["predeclared_direction_and_recovery_checks"][1]["direction_status"])

        opposite = module.paired_response(control, arm(-1.0, 7.0, 40.0, 95.0, 8.0, 1.0))
        self.assertFalse(opposite["predeclared_direction_and_recovery_checks"][1]["passed"])
        self.assertFalse(opposite["predeclared_direction_and_recovery_checks"][2]["passed"])
        self.assertIn("opposite", opposite["predeclared_direction_and_recovery_checks"][1]["direction_status"])

    def test_float32_midpoint_cells_cover_binades_zero_and_subnormals(self):
        center, lower, upper = module.float32_rounding_cell("1.0")
        self.assertEqual(center, Fraction(1, 1))
        self.assertEqual(lower, Fraction(1, 1) - Fraction(1, 2**25))
        self.assertEqual(upper, Fraction(1, 1) + Fraction(1, 2**24))

        center, lower, upper = module.float32_rounding_cell("0.0")
        self.assertEqual(center, Fraction(0, 1))
        self.assertEqual(lower, -Fraction(1, 2**150))
        self.assertEqual(upper, Fraction(1, 2**150))

        min_subnormal = format(2.0**-149, ".12g")
        center, lower, upper = module.float32_rounding_cell(min_subnormal)
        self.assertEqual(center, Fraction(1, 2**149))
        self.assertEqual(lower, Fraction(1, 2**150))
        self.assertEqual(upper, Fraction(3, 2**150))

            "verified": True,
            "expected_path": module.GAS_RUNTIME_DYLIB,

            "runtime_source_head": module.GAS_RUNTIME_SOURCE_HEAD,

        self.assertTrue(any("invocation" in reason for reason in result["reasons_unverified"]))
        facts = self.facts()
        del facts["declaration_assets"][module.GAS_RESPIRATION_METALLIB]
        result = self.classify(facts)
        self.assertFalse(result["identity_verified"])

    def test_wrong_loaded_dylib_or_source_hash_fails_closed(self):
        facts = self.facts()
        facts["metadata"]["loaded_metal_runtime"]["expected_sha256"] = "0" * 64
        self.assertFalse(self.classify(facts)["identity_verified"])
        facts = self.facts()
        facts["source_hashes"][str(module.GAS_METAL_SOURCE)] = "0" * 64
        self.assertFalse(self.classify(facts)["identity_verified"])

    def test_build_rule_must_bind_source_to_shader_target(self):
        facts = self.facts()
        facts["build_rule_matches_source"] = False
        self.assertFalse(self.classify(facts)["identity_verified"])


class RuntimeGuardStatusTests(unittest.TestCase):
    def test_guard_pass_is_separate_from_direct_residual_availability(self):
        self.assertTrue(module.accepted_runtime_guard_passed(True, module.STEPS))
        self.assertFalse(module.accepted_runtime_guard_passed(False, module.STEPS))
        self.assertFalse(module.accepted_runtime_guard_passed(True, module.STEPS - 1))

    def test_identity_match_alone_does_not_pass_incomplete_horizon(self):
        self.assertFalse(module.accepted_runtime_guard_passed(True, 0))
        self.assertFalse(module.accepted_runtime_guard_passed(1, module.STEPS))
