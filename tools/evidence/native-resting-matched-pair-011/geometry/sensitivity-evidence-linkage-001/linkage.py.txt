LIMIT_STEP = 30000
DT_F32 = struct.unpack("<f", struct.pack("<f", 0.002))[0]
CLOCK_END = LIMIT_STEP * DT_F32
pin_map: dict[str, str] = {}

def sha(path: Path) -> str:
    h = hashlib.sha256()
    with path.open("rb") as f:
        for block in iter(lambda: f.read(4 * 1024 * 1024), b""):
            h.update(block)
    return h.hexdigest()

def pin(path: Path) -> str:
    if not path.is_file():
        raise RuntimeError(f"missing required input: {path}")
    value = sha(path)
    pin_map[str(path)] = value
    return value

def read_json(path: Path):
    pin(path)
    return json.loads(path.read_text())

def require(condition: bool, message: str):
    if not condition:
        raise RuntimeError(message)

def native_configs(meta: dict) -> tuple[Path, Path]:
    argv = meta["argv"]
    i = argv.index("--resting-scene")
    require(i + 2 < len(argv), "native argv lacks paired circulation/respiration paths")
    return Path(argv[i + 1]), Path(argv[i + 2])

def numeric_leaves(obj, prefix=""):
    result = {}
    if isinstance(obj, dict):
        for k, v in obj.items():
            result.update(numeric_leaves(v, f"{prefix}/{k}"))
    elif isinstance(obj, list):
        for i, v in enumerate(obj):
            result.update(numeric_leaves(v, f"{prefix}/{i}"))
    elif isinstance(obj, (int, float)) and not isinstance(obj, bool):
        result[prefix or "/"] = obj
    return result

def csv_prefix(path: Path, step_column: str, extra_key=()):
    pin(path)
    with path.open(newline="") as f:
        reader = csv.DictReader(f)
        header = list(reader.fieldnames or [])
        require(step_column in header, f"{path}: missing {step_column}")
        for k in extra_key:
            require(k in header, f"{path}: missing key column {k}")
        keys = {}
        rows = 0
        for row in reader:
            step = int(row[step_column])
            if step > LIMIT_STEP:
                continue
            key = (step,) + tuple(int(row[k]) for k in extra_key)
            require(key not in keys, f"{path}: duplicate row key {key}")
            keys[key] = row
            rows += 1
    ordered = sorted(keys)
    require(ordered == list(keys.keys()) or len(keys) < 2, f"{path}: prefix rows are not ordered by step/key")
    return header, keys, rows

def compare_csv(name: str, step_column: str, extra_key=()):
    left_path = SBASE / name
    right_path = PBASE / name
    h0, left, n0 = csv_prefix(left_path, step_column, extra_key)
    h1, right, n1 = csv_prefix(right_path, step_column, extra_key)
    require(h0 == h1, f"{name}: header differs between runs")
    left_keys, right_keys = set(left), set(right)
    shared = left_keys & right_keys
    missing_left = sorted(right_keys - left_keys)
    missing_right = sorted(left_keys - right_keys)
    changed = []
    for key in sorted(shared):
        for col in h0:
            if left[key][col] != right[key][col]:
                changed.append({
                    "key": list(key), "column": col,
                    "sensitivity_token": left[key][col],
                    "paired011_token": right[key][col],
                })
                if len(changed) >= 20:
                    break
        if len(changed) >= 20:
            break
    return {
        "sensitivity_path": str(left_path), "paired011_path": str(right_path),
        "sensitivity_sha256": pin_map[str(left_path)],
        "paired011_sha256": pin_map[str(right_path)],
        "header_equal": h0 == h1, "column_count": len(h0),
        "prefix_row_count_sensitivity": n0, "prefix_row_count_paired011": n1,
        "key_count_sensitivity": len(left), "key_count_paired011": len(right),
        "shared_key_count": len(shared),
        "only_in_sensitivity_count": len(missing_right),
        "only_in_paired011_count": len(missing_left),
        "first_only_in_sensitivity_keys": [list(x) for x in missing_right[:8]],
        "first_only_in_paired011_keys": [list(x) for x in missing_left[:8]],
        "all_shared_cells_exact": not changed and not missing_left and not missing_right,
        "changed_cell_count_lower_bound": len(changed),
        "first_changed_cells": changed,
    }

def selected_respiration(resp):
    keys = [
        "oxygen_consumption_stpd_m3_per_s",
        "carbon_dioxide_production_stpd_m3_per_s",
        "frc_m3",
        "lung_compliance_m3_per_pa",
        "airway_resistance_pa_s_per_m3",
    ]
    out = {k: resp.get(k) for k in keys}
    brain = resp.get("brain_parameters", {})
    out["brain_parameters"] = {
        k: brain.get(k) for k in (
            "CO2ResponseSlopeLitresPerMinutePerMmHg",
            "centralCO2Fraction",
            "centralTimeConstantSeconds",
            "peripheralTimeConstantSeconds",
            "restingFrequencyBreathsPerMinute",
            "restingMinuteVentilationLitresPerMinute",
            "restingTidalVolumeLitres",
        )
    }
    return out

def runtime_assets(meta):
    assets = meta.get("asset_sha256", {})
    names = (
        "numi-human-native", "libmetalrobo.dylib",
        "HumanRespiration.metallib", "NumiMatter.metallib",
        "NumiMatterPhysicalStateDigest.metallib", "MetalRobo.metallib",
        "MetalRoboHyperPolicy.metallib", "NumiNeuron.metallib",
    )
    return {
        name: [{"path": path, "sha256": value} for path, value in sorted(assets.items())
               if Path(path).name == name]
        for name in names
    }

    for arm in ("metabolism-joint-0p8", "metabolism-joint-1p2",
                "co2-response-slope-0p8", "co2-response-slope-1p2"):

        require(exe.get("postflight", {}).get("checks", {}).get("argv_matches_source_owner") is True,

        require(rep.get("status") == "closed_descriptive_sensitivity_comparison"
                and grid.get("last_step") == 30000
                and abs(grid.get("last_time_s", 0) - CLOCK_END) < 1e-10
                and rep.get("accepted_grid", {}).get("same_step_and_actual_time_tokens") is True,

            "analysis_path": str(rep_path), "analysis_sha256": pin_map[str(rep_path)],

            "accepted_duration_s": grid["last_time_s"],

            "baseline original execution does not record native rc0")
    require(fail.get("postflight", {}).get("checks", {}).get("argv_matches_preview") is False,

    runtime_pair = runtime_assets(pair_meta)

    runtime_same = {}

        b = runtime_pair[name]
        if a and b and [x["sha256"] for x in a] == [x["sha256"] for x in b]:
            runtime_same[name] = {"sensitivity": a, "paired011": b}
        elif a or b:
            runtime_diff[name] = {"sensitivity": a, "paired011": b}

    report = {
