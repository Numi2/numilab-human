cmake_minimum_required(VERSION 3.16)
project(febio_public_reference LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(OPENMP_ROOT "/opt/homebrew/opt/libomp" CACHE PATH "Existing OpenMP installation")
set(SUPERLU_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../superlu-a8a0caafa593772450b35339c8fa4e76b7bc934f")
set(SUPERLU_LIBRARY "${CMAKE_CURRENT_SOURCE_DIR}/../superlu-build/SRC/libsuperlu.a")
file(GLOB sources CONFIGURE_DEPENDS FEBioLib/*.cpp FEBioMech/*.cpp FEBioMix/*.cpp FEBioFluid/*.cpp FEBioOpt/*.cpp FEBioPlot/*.cpp FEBioTest/*.cpp FEBioXML/*.cpp FECore/*.cpp NumCore/*.cpp)
list(FILTER sources EXCLUDE REGEX "(BIPNSolver|FEASTEigenSolver).cpp$")
add_library(febio_reference_objects OBJECT ${sources})
target_include_directories(febio_reference_objects PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} "${OPENMP_ROOT}/include" "${SUPERLU_ROOT}/SRC")
target_compile_definitions(febio_reference_objects PUBLIC HAVE_ZLIB SUPERLU)
target_compile_options(febio_reference_objects PUBLIC -Xpreprocessor -fopenmp -Wno-deprecated-declarations -Wno-inconsistent-missing-override)
target_link_libraries(febio_reference_objects PUBLIC "${OPENMP_ROOT}/lib/libomp.dylib" "${SUPERLU_LIBRARY}" "-framework Accelerate" z)
file(GLOB main_sources CONFIGURE_DEPENDS FEBio3/*.cpp)
add_executable(febio3 ${main_sources})
target_link_libraries(febio3 PRIVATE febio_reference_objects)
add_executable(febio-material-oracle OpenKneeMaterialOracle.cpp)
target_link_libraries(febio-material-oracle PRIVATE febio_reference_objects)
